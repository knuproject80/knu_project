# frontend-ui/.env.production  (또는 Vercel 대시보드 Environment Variables에 입력)

# Spring WebSocket — 반드시 wss:// (https 페이지에서 ws:// 는 브라우저가 차단)
VITE_WS_URL=wss://<backend-spring 공개도메인>/ws

# 프론트 이벤트 발행 경로 — WebSocketConfig의 클라이언트→서버 prefix가 /app 이므로
# StompRelayController 중계를 타려면 /app 경로로 보내야 함 (기본값 /topic 아님)
VITE_FRONT_EVENTS_DESTINATION=/app/front/events
VITE_FRONT_ACK_DESTINATION=/app/front/ack
